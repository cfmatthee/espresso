[dependencies]
tray-icon = "0.26"
image = { version = "0.25", default-features = false, features = ["png"] }




use std::{thread, time::Duration};
use image::ImageReader;
use tray_icon::{Icon, TrayIcon, TrayIconBuilder};

#[derive(Clone, Copy)]
enum AppState {
    Idle,
    Active,
}

fn load_icon(path: &str) -> Icon {
    let image = ImageReader::open(path)
        .expect("Failed to open icon")
        .decode()
        .expect("Failed to decode icon")
        .into_rgba8();

    let (width, height) = image.dimensions();

    Icon::from_rgba(image.into_raw(), width, height)
        .expect("Invalid icon image")
}

fn main() {
    let empty = load_icon("assets/espresso_empty_32.png");
    let filled = load_icon("assets/espresso_filled_32.png");

    let tray = TrayIconBuilder::new()
        .with_icon(empty.clone())
        .with_tooltip("Espresso Tracker — Idle")
        .build()
        .expect("Failed to create tray icon");

    let mut state = AppState::Idle;

    // Demonstration only: replace this with your real application events.
    loop {
        thread::sleep(Duration::from_secs(5));

        state = match state {
            AppState::Idle => AppState::Active,
            AppState::Active => AppState::Idle,
        };

        let (icon, tooltip) = match state {
            AppState::Idle => (&empty, "Espresso Tracker — Idle"),
            AppState::Active => (&filled, "Espresso Tracker — Active"),
        };

        tray.set_icon(Some(icon.clone()))
            .expect("Failed to update tray icon");

        tray.set_tooltip(Some(tooltip))
            .expect("Failed to update tray tooltip");
    }
}
